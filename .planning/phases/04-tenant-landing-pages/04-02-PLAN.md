    import type { ReactNode } from 'react';
    import type { TenantConfig, ContactInfo, SocialLink } from './tenant-types';
    import { Facebook, Twitter, Instagram, Linkedin } from 'lucide-react';

    interface FooterProps {
      tenant: TenantConfig;
      contact?: ContactInfo;
      socialLinks?: SocialLink[];
      navLinks?: { label: string; href: string }[];
    }

    const socialIcons = { facebook: <Facebook size={16} />, twitter: <Twitter size={16} />, instagram: <Instagram size={16} />, linkedin: <Linkedin size={16} /> };

    export function Footer({ tenant, contact, socialLinks, navLinks }: FooterProps) {
      const isEmerald = tenant.theme === 'emerald';
      return (
        <footer className="border-t border-separator bg-gray-50">
          <div className="mx-auto max-w-5xl px-4 py-12">
            <div className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-4 gap-8">
              <div className="space-y-3">
                <div className="flex items-center gap-2">
                  <div className={`flex h-8 w-8 items-center justify-center rounded-lg ${isEmerald ? 'bg-emerald-600' : 'bg-violet-600'} text-white text-xs font-bold`}>{tenant.shortName}</div>
                  <span className="text-sm font-semibold text-foreground">{tenant.name}</span>
                </div>
                <p className="text-xs text-muted">Tracking across {tenant.name === 'Gajan Traders' ? 'India' : 'Sri Lanka'}.</p>
              </div>
              {navLinks?.length > 0 && (
                <div className="space-y-3">
                  <h3 className="text-xs font-semibold uppercase tracking-wider">Quick Links</h3>
                  <ul className="space-y-2">{navLinks.map(l => <li key={l.href}><a href={l.href} className="text-xs text-muted hover:text-foreground">{l.label}</a></li>)}</ul>
                </div>
              )}
              {contact && (
                <div className="space-y-3">
                  <h3 className="text-xs font-semibold uppercase tracking-wider">Contact</h3>
                  <ul className="space-y-2 text-xs text-muted">
                    {contact.address && <li>{contact.address}</li>}
                    {contact.phone && <li>{contact.phone}</li>}
                    {contact.email && <li>{contact.email}</li>}
                  </ul>
                </div>
              )}
              {socialLinks?.length > 0 && (
                <div className="space-y-3">
                  <h3 className="text-xs font-semibold uppercase tracking-wider">Follow Us</h3>
                  <div className="flex items-center gap-3">
                    {socialLinks.map(s => (
                      <a key={s.platform} href={s.url} target="_blank" rel="noopener noreferrer"
                        className={`text-muted hover:${isEmerald ? 'text-emerald-600' : 'text-violet-600'}`}
                        aria-label={s.platform}>{socialIcons[s.platform]}</a>
                    ))}
                  </div>
                </div>
              )}
            </div>
          </div>
          <div className="border-t border-separator py-4">
            <div className="mx-auto max-w-5xl px-4 text-center text-xs text-muted">
              &copy; {new Date().getFullYear()} {tenant.name}. All rights reserved.
            </div>
          </div>
        </footer>
      );
    }
