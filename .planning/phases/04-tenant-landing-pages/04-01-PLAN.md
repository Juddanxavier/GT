    ```typescript
    'use client';

    import type { ReactNode } from 'react';
    import { Navbar } from '@repo/ui/navbar';
    import { useTenantConfig } from '@/lib/use-tenant-config';
    import { authClient } from '@/lib/auth-client';
    import { useUnreadCount } from '@/lib/queries';
    import { Package, FileText, Receipt, Bell, User, LogOut, LogIn, UserPlus } from 'lucide-react';
    import type { NavLink } from '@repo/ui/tenant-types';

    const loggedInLinks: NavLink[] = [
      { label: 'My Shipments', href: '/my-shipments', icon: 'Package', match: ['/my-shipments'] },
      { label: 'Quotations', href: '/my-quotations', icon: 'FileText', match: ['/my-quotations', '/quotations'] },
      { label: 'Invoices', href: '/my-invoices', icon: 'Receipt', match: ['/my-invoices', '/invoices'] },
      { label: 'Notifications', href: '/notifications', icon: 'Bell', match: ['/notifications'] },
      { label: 'Profile', href: '/profile', icon: 'User', match: ['/profile'] },
    ];

    const loggedOutLinks: NavLink[] = [
      { label: 'Sign in', href: '/login', icon: 'LogIn' },
      { label: 'Register', href: '/register', icon: 'UserPlus' },
    ];

    export default function PortalLayout({ children }: { children: ReactNode }): ReactNode {
      const config = useTenantConfig();
      const { data: session } = authClient.useSession();
      const { data: unreadData } = useUnreadCount();

      return (
        <>
          <Navbar
            tenant={config}
            links={{ loggedIn: loggedInLinks, loggedOut: loggedOutLinks }}
            session={session ?? null}
            unreadCount={unreadData?.count ?? 0}
            onSignOut={async () => { await authClient.signOut(); }}
          />
          <main className="mx-auto max-w-5xl px-4 py-8">{children}</main>
        </>
      );
    }
    ```
