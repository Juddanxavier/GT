    - Create a helper:
      ```typescript
      import { createHash } from 'node:crypto';

      function eventDedupKey(event: NormalizedEvent): { eventTime: Date; subStatus: string } {
        if (event.subStatus && event.subStatus.trim().length > 0) {
          return { eventTime: event.eventTime, subStatus: event.subStatus };
        }
        // Fallback: hash the description for dedup when subStatus is unavailable
        const hash = createHash('sha256')
          .update(`${event.eventTime.toISOString()}:${event.description}:${event.location}`)
          .digest('hex')
          .slice(0, 32);
        return { eventTime: event.eventTime, subStatus: `hash:${hash}` };
      }
      ```
