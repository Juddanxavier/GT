    ```typescript
    'use client';
    import type { ReactNode } from 'react';
    import { STATUS_META } from '@repo/ui/status-config';

    interface TrackingTimelineProps {
      events: Array<{
        status: string;
        description: string;
        location: string;
        eventTime: string;
      }>;
    }

    export function TrackingTimeline({ events }: TrackingTimelineProps): ReactNode {
      if (!events?.length) {
        return <div className="text-muted text-sm py-8 text-center">No tracking events yet</div>;
      }

      return (
        <div className="space-y-0">
          {events.map((event, idx) => {
            const meta = STATUS_META[event.status as keyof typeof STATUS_META];
            const isLatest = idx === 0;
            return (
              <div key={idx} className="flex gap-4">
                {/* Timeline dot + line */}
                <div className="flex flex-col items-center">
                  <div className={`w-3 h-3 rounded-full ${isLatest ? 'ring-2 ring-offset-2' : ''}`}
                    style={{ backgroundColor: meta?.color === 'success' ? '#22c55e' : meta?.color === 'danger' ? '#ef4444' : meta?.color === 'primary' ? '#3b82f6' : '#9ca3af' }} />
                  {idx < events.length - 1 && <div className="w-0.5 flex-1 bg-border" />}
                </div>
                {/* Event content */}
                <div className={`pb-6 flex-1 ${isLatest ? '' : 'opacity-70'}`}>
                  <p className="text-sm font-medium">{meta?.label ?? event.status}</p>
                  {event.description && <p className="text-xs text-muted">{event.description}</p>}
                  {event.location && <p className="text-xs text-muted flex items-center gap-1"><MapPin size={12} />{event.location}</p>}
                  <p className="text-xs text-muted mt-1">
                    {new Date(event.eventTime).toLocaleString()}
                    {idx > 0 && ` (${elapsedSince(events[idx - 1].eventTime, event.eventTime)})`}
                  </p>
                </div>
              </div>
            );
          })}
        </div>
      );
    }

    function elapsedSince(prev: string, curr: string): string {
      const diff = new Date(curr).getTime() - new Date(prev).getTime();
      const hours = Math.floor(diff / 3600000);
      const mins = Math.floor((diff % 3600000) / 60000);
      if (hours > 0) return `${hours}h ${mins}m`;
      return `${mins}m`;
    }
    ```

    2. **Create `apps/api/src/services/notification-prefs.ts`**:
    ```typescript
    import { and, eq } from 'drizzle-orm';
    import { db } from '../db';
    import { notificationPreferences } from '../db/schema';

    type PrefCategory = 'delivery' | 'exception' | 'general';
    type PrefChannel = 'email' | 'whatsapp';

    const categoryChannelMap: Record<PrefCategory, { email: keyof typeof notificationPreferences.$inferInsert; whatsapp: keyof typeof notificationPreferences.$inferInsert }> = {
      delivery: { email: 'deliveryEmail', whatsapp: 'deliveryWhatsapp' },
      exception: { email: 'exceptionEmail', whatsapp: 'exceptionWhatsapp' },
      general: { email: 'generalEmail', whatsapp: 'generalWhatsapp' },
    };

    export async function getPreferences(userId: string, tenantId: string) {
      const [prefs] = await db
        .select()
        .from(notificationPreferences)
        .where(and(
          eq(notificationPreferences.userId, userId),
          eq(notificationPreferences.tenantId, tenantId),
        ))
        .limit(1);

      return prefs ?? null;
    }

    export async function upsertPreferences(
      userId: string,
      tenantId: string,
      prefs: Partial<typeof notificationPreferences.$inferInsert>,
    ) {
      const existing = await getPreferences(userId, tenantId);

      if (existing) {
        const [updated] = await db
          .update(notificationPreferences)
          .set({ ...prefs, updatedAt: new Date() })
          .where(eq(notificationPreferences.id, existing.id))
          .returning();
        return updated;
      }

      const [created] = await db
        .insert(notificationPreferences)
        .values({
          userId,
          tenantId,
          deliveryEmail: true,
          deliveryWhatsapp: true,
          exceptionEmail: true,
          exceptionWhatsapp: true,
          generalEmail: true,
          generalWhatsapp: true,
          ...prefs,
        })
        .returning();
      return created;
    }

    export async function shouldNotifyByWhatsapp(
      userId: string,
      tenantId: string,
      category: PrefCategory,
    ): Promise<boolean> {
      const prefs = await getPreferences(userId, tenantId);
      if (!prefs) return true; // default: all on
      const channelKey = categoryChannelMap[category]?.whatsapp;
      return channelKey ? Boolean(prefs[channelKey]) : true;
    }

    export async function shouldNotifyByEmail(
      userId: string,
      tenantId: string,
      category: PrefCategory,
    ): Promise<boolean> {
      const prefs = await getPreferences(userId, tenantId);
      if (!prefs) return true;
      const channelKey = categoryChannelMap[category]?.email;
      return channelKey ? Boolean(prefs[channelKey]) : true;
    }
    ```

    ```typescript
              <SelectItem key="status_update">Status Update</SelectItem>

            )}
            {notifications.map((n: { id: string; type: string; title: string; message: string; read: boolean; createdAt: string }) => (

                <CardBody className="flex items-start gap-3">
                  <Checkbox isSelected={selected.has(n.id)} onValueChange={(v) => {
                    const next = new Set(selected);
                    v ? next.add(n.id) : next.delete(n.id);
                    setSelected(next);
                  }} />
                  <div className="flex-1 min-w-0">
                    <p className="text-sm font-medium">{n.title}</p>
                    <p className="text-xs text-muted mt-1">{n.message}</p>
                    <p className="text-xs text-muted mt-1">{new Date(n.createdAt).toLocaleString()}</p>
                  </div>
                  {!n.read && <Chip size="sm" color="primary" variant="flat">New</Chip>}
                </CardBody>
              </Card>
            ))}
          </div>

          {totalPages > 1 && (
            <div className="flex justify-center">
              <Pagination total={totalPages} page={page} onChange={setPage} />
            </div>
          )}
        </div>
      );
    }
    ```

    ```typescript
        { key: 'general', label: 'General Updates', desc: 'Account notifications, promotional' },
      ];

      return (
        <div className="space-y-4 max-w-2xl">
          <h1 className="text-xl font-semibold">Notification Preferences</h1>
          {categories.map(cat => (
            <Card key={cat.key}>

              <CardBody className="space-y-4">
                <div className="flex items-center justify-between">
                  <div className="flex items-center gap-2"><Mail size={16} /><span className="text-sm">Email</span></div>

                  <div className="flex items-center gap-2"><MessageSquare size={16} /><span className="text-sm">WhatsApp</span></div>
                  <Switch isSelected={prefs?.[`${cat.key}Whatsapp`]} onValueChange={() => toggle(`${cat.key}Whatsapp`)} />
                </div>
                <p className="text-xs text-muted">{cat.desc}</p>
              </CardBody>
            </Card>
          ))}
          <Button color="primary" onPress={save} isLoading={saving}>Save Preferences</Button>
        </div>
      );
    }
    ```
