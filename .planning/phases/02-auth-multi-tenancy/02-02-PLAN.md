       ```typescript
       /** @format */
       
       import { betterAuth } from 'better-auth';
       import { drizzleAdapter } from '@better-auth/drizzle-adapter';
       import { db } from './db';
       import * as schema from './db/schema';
       
       export const auth = betterAuth({
         database: drizzleAdapter(db, {
           provider: 'pg',
           schema: {
             ...schema,
             user: schema.user,
             session: schema.session,
             account: schema.account,
             verification: schema.verification,
           },
         }),
         emailAndPassword: {
           enabled: true,
           autoSignIn: true,
         },
         socialProviders: {
           google: {
             clientId: process.env.GOOGLE_CLIENT_ID ?? '',
             clientSecret: process.env.GOOGLE_CLIENT_SECRET ?? '',
           },
         },
         user: {
           additionalFields: {
             tenantId: {
               type: 'string',
               required: false,
               input: false,
             },
             role: {
               type: 'string',
               required: true,
               defaultValue: 'customer',
               input: false,
             },
             branchId: {
               type: 'string',
               required: false,
               input: false,
             },
           },
         },
         advanced: {
           crossSubDomainCookies: {
             enabled: true,
           },
           defaultCookieAttributes: {
             sameSite: 'lax',
             secure: false, // Set to true in production with HTTPS
           },
         },
         // CSRF protection is enabled by default
         // Session refresh rotation is enabled by default (1/3 through lifetime)
       });
       
       export type Auth = typeof auth;
       ```

       **Critical notes:**
       - `additionalFields` must match the Drizzle schema column names. The `input: false` prevents users from setting these fields during signup — they are assigned by the server.
       - `crossSubDomainCookies.enabled: true` allows cookies to work across subdomains (admin.gajantraders.com → api.gajantraders.com).
       - `secure: false` is for localhost development. Must be changed to `true` in production.
       - Drizzle adapter schema mapping must include ALL tables the adapter should be aware of. Pass the schema object with explicit mappings.

    4. **Verify the auth module loads:**

       ```bash
       node -e "require('tsx').register(); require('./apps/api/src/auth')"
       ```
       This should not throw any errors. If it does, it's likely a Better Auth field mismatch between schema.ts and additionalFields.

    **Why this order:** Install first, then configure. The auth config references the Drizzle client which already exists from Phase 1.

  <name>Task 2: Create Hono auth middleware and mount Better Auth handler</name>
  <files>apps/api/src/index.ts, apps/api/src/middleware/auth.ts</files>
  <action>
    1. **Create `apps/api/src/middleware/` directory** (create the folder):

    2. **Create `apps/api/src/middleware/auth.ts`** — Session middleware:

       ```typescript
       /** @format */
       
       import type { Context, Next } from 'hono';
       import { auth } from '../auth';
       
       // Extend Hono's context variables type
       export type AuthVariables = {
         user: typeof auth.$Infer.Session.user | null;
         session: typeof auth.$Infer.Session.session | null;
       };
       
       /**
        * Middleware that fetches the current session and injects user + session
        * into Hono context variables for downstream route handlers.
        * 
        * - If the request has a valid session cookie/token, user and session 
        *   are populated.
        * - If not, both are set to null (requests proceed without auth).
        * - Routes requiring authentication should check c.get('user') and
        *   return 401 if null.
        */
       export async function sessionMiddleware(c: Context, next: Next) {
         const session = await auth.api.getSession({
           headers: c.req.raw.headers,
         });
         
         if (!session) {
           c.set('user', null);
           c.set('session', null);
           await next();
           return;
         }
         
         c.set('user', session.user);
         c.set('session', session.session);
         await next();
       }
       ```

    3. **Update `apps/api/src/index.ts`** — Mount auth handler with CORS:

       ```typescript
       /** @format */
       
       import { serve } from '@hono/node-server';
       import { Hono } from 'hono';
       import { cors } from 'hono/cors';
       import { auth } from './auth';
       import { sessionMiddleware, type AuthVariables } from './middleware/auth';
       
       const app = new Hono<{
         Variables: AuthVariables;
       }>();
       
       // CORS — must be before auth handler for preflight requests
       app.use(
         '/api/auth/*',
         cors({
           origin: ['http://localhost:3000'], // Admin app dev URL
           allowHeaders: ['Content-Type', 'Authorization'],
           allowMethods: ['POST', 'GET', 'OPTIONS'],
           exposeHeaders: ['Content-Length'],
           maxAge: 600,
           credentials: true,
         }),
       );
       
       // CORS for all other API routes
       app.use('*', cors({
         origin: ['http://localhost:3000'],
         credentials: true,
       }));
       
       // Mount Better Auth handler
       app.on(['POST', 'GET'], '/api/auth/*', (c) => {
         return auth.handler(c.req.raw);
       });
       
       // Session middleware for all routes (including after auth handler)
       app.use('*', sessionMiddleware);
       
       // Health check
       app.get('/', (c) => {
         return c.json({ message: 'ShipTrack API' });
       });
       
       // Protected route example (commented — for reference)
       // app.get('/api/me', (c) => {
       //   const user = c.get('user');
       //   if (!user) return c.json({ error: 'Unauthorized' }, 401);
       //   return c.json({ user });
       // });
       
       serve({
         fetch: app.fetch,
         port: Number(process.env.PORT) || 3001,
       });
       ```

       **Critical notes:**
       - CORS must be registered BEFORE the auth handler. Better Auth uses POST for login/signup which triggers preflight OPTIONS requests. Without CORS first, the preflight fails.
       - The `auth.handler(c.req.raw)` takes the raw Request, not the Hono context. This is because Better Auth needs the raw headers for cookie parsing.
       - Session middleware runs AFTER the auth handler (the auth handler manages its own session creation). Routes registered before the session middleware (like the auth handler itself) don't need it.
       - The `Hono<{ Variables: AuthVariables }>` type parameter ensures `c.get('user')` and `c.get('session')` are properly typed.

## Threat Model

### Trust Boundaries
| Boundary | Description |
|----------|-------------|
| `Browser ←→ Hono API` | Session cookies and API tokens cross network boundary |
| `Better Auth → NeonDB` | Session, user, account data read/written to cloud database |

### STRIDE Threat Register
| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-05 | Spoofing | Session cookie | mitigate | Better Auth uses signed cookies with httpOnly flag. CSRF protection enabled by default. SameSite=Lax applied. |
| T-02-06 | Tampering | Auth token | mitigate | Better Auth implements refresh token rotation — each refresh invalidates the previous token. Access tokens (session cookies) are rotated at 1/3 of lifetime. |
| T-02-07 | Information Disclosure | Google OAuth callback | accept | Google OAuth client ID/secret are placeholders in Phase 2. Production setup requires HTTPS (OAuth redirect_uri must be HTTPS). |
| T-02-08 | Elevation of Privilege | Custom fields | mitigate | `additionalFields` have `input: false` — users cannot set tenantId, role, or branchId during signup. Only auth.ts or superadmin API can set these. |
| T-02-09 | Repudiation | Session creation | accept | Audit logging (Phase 3+) will add session creation traceability. |
| T-02-10 | Information Disclosure | CORS | mitigate | CORS origin restricted to `http://localhost:3000` (admin app). Production will restrict to specific domains. |
